void loop() {
  if (!Serial.available()) return;

  String line = Serial.readStringUntil('\n');
  line.trim();
  if (line.length() == 0) return;

  if (line.equalsIgnoreCase("STAND")) {
    applyPose(STAND);
    isStanding = true;
    Serial.println(F("-> STAND pose applied (wheels enabled)"));

  } else if (line.equalsIgnoreCase("SIT")) {
    wheelsStop();
    applyPose(SIT);
    isStanding = false;
    Serial.println(F("-> SIT pose applied (wheels disabled)"));

  } else if (line.equalsIgnoreCase("PRINT")) {
    for (int ch = 0; ch < 8; ch++) {
      Serial.print(F("CH")); Serial.print(ch);
      Serial.print(F(": ")); Serial.println(currentAngle[ch]);
    }

  } else if (line.equalsIgnoreCase("F")) {
    if (isStanding) { wheelsForward(); Serial.println(F("-> wheels forward")); }
    else Serial.println(F("Ignored: not standing"));

  } else if (line.equalsIgnoreCase("B")) {
    if (isStanding) { wheelsBackward(); Serial.println(F("-> wheels backward")); }
    else Serial.println(F("Ignored: not standing"));

  } else if (line.equalsIgnoreCase("S")) {
    wheelsStop();
    Serial.println(F("-> wheels stopped"));

  } else if (line.equalsIgnoreCase("L")) {
    if (isStanding) { wheelsTurnLeft(); Serial.println(F("-> turning left")); }
    else Serial.println(F("Ignored: not standing"));

  } else if (line.equalsIgnoreCase("R")) {
    if (isStanding) { wheelsTurnRight(); Serial.println(F("-> turning right")); }
    else Serial.println(F("Ignored: not standing"));

  } else {
    int spaceIdx = line.indexOf(' ');
    if (spaceIdx > 0) {
      int ch  = line.substring(0, spaceIdx).toInt();
      int deg = line.substring(spaceIdx + 1).toInt();
      if (ch >= 0 && ch <= 7) {
        setAngle(ch, deg);
        Serial.print(F("CH")); Serial.print(ch);
        Serial.print(F(" -> ")); Serial.println(deg);
      } else {
        Serial.println(F("Invalid channel (0-7)"));
      }
    } else {
      Serial.println(F("Unrecognized command"));
    }
  }
}
